#!/usr/bin/env python3
# ISP 1 lab. — Cezario šifras
# 1. Užšifruoti su duotu poslinkiu.  2. Atšifruoti (poslinkis randamas dažnine analize).
import math, re, sys
from pathlib import Path

ABC = "aąbcčdeęėfghiįyjklmnoprsštuųūvzž"
N, UP = len(ABC), ABC.upper()
IDX = {c: i for i, c in enumerate(ABC)} | {c: i for i, c in enumerate(UP)}
DAZNIS = {'i':11,'a':10.5,'s':7,'o':5.2,'t':5,'n':4.8,'e':4.5,'r':4.3,'k':4.2,'u':4,
          'm':3.3,'l':3.2,'p':3,'d':2.8,'v':2.3,'j':2.2,'g':1.7,'b':1.6,'ė':1.6,'ž':1.3,
          'y':1.2,'š':1.2,'č':1,'ą':1,'z':0.9,'ų':0.8,'ū':0.8,'ę':0.6,'į':0.6,'c':0.6}
LOGP = {c: math.log(DAZNIS.get(c, 0.05) / 100) for c in ABC}  # log-tikimybės (su grindimis)

def poslinkis(t, k):
    # Pastumia abėcėlės raides per k; kiti simboliai nekeičiami, registras išsaugomas.
    return "".join((ABC if c.islower() else UP)[(IDX[c]+k) % N] if c in IDX else c for c in t)

def rasti_rakta(sifras):
    # Poslinkis, kuris atšifruoja į labiausiai lietuvišką tekstą.
    balas = lambda t: sum(LOGP.get(c, 0) for c in t.lower())
    return max(range(N), key=lambda k: balas(poslinkis(sifras, -k)))

def spresti(v):
    sekcija = uzsif = atsif = None
    for l in Path("variantai.txt").read_text(encoding="utf-8").splitlines():
        l = l.strip()
        if l in ("Užšifruoti", "Atšifruoti"):
            sekcija = l; continue
        m = re.match(rf"^{v}\.\s+(.*)$", l)
        if not m: continue
        if sekcija == "Užšifruoti":
            tekstas, _, k = m.group(1).rpartition(" ")
            uzsif = poslinkis(tekstas, int(k))
        elif sekcija == "Atšifruoti":
            atsif = poslinkis(m.group(1), -rasti_rakta(m.group(1)))
    return (f"Variantas: {v}\n\n1. Užšifruotas tekstas:\n   {uzsif}\n\n"
            f"2. Atšifruotas tekstas:\n   {atsif}\n")

if __name__ == "__main__":
    print(spresti(int(sys.argv[1])))
