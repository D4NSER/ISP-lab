Variantas: 5

1. Atšifruotas tekstas
   Šifruotas tekstas: Mrėszjčs ęįškhrėaf ėy gšįzžzjzs sryif lyabūfa.
   Duotas raktas: mūša
   Rezultatas: Auksinis plaktukas ir geležines duris prakala.

2. Raktas
   Rastas raktas: viešasisraktas
   (Rakto ilgis 14 nustatytas sutapimo indeksu (IC); raidės — dažnine analize.)

Programos kodas (vigenere.py):
----------------------------------------------------------------------
#!/usr/bin/env python3
# ISP 2 lab. — Viženerio šifras
# 1. Atšifruoti su duotu raktu.  2. Rasti raktą (kriptoanalizė: IC + dažnine analize).
import math, sys
from pathlib import Path

ABC = "aąbcčdeęėfghiįyjklmnoprsštuųūvzž"
N, UP = len(ABC), ABC.upper()
IDX = {c: i for i, c in enumerate(ABC)} | {c: i for i, c in enumerate(UP)}
DAZNIS = {'i':11,'a':10.5,'s':7,'o':5.2,'t':5,'n':4.8,'e':4.5,'r':4.3,'k':4.2,'u':4,
          'm':3.3,'l':3.2,'p':3,'d':2.8,'v':2.3,'j':2.2,'g':1.7,'b':1.6,'ė':1.6,'ž':1.3,
          'y':1.2,'š':1.2,'č':1,'ą':1,'z':0.9,'ų':0.8,'ū':0.8,'ę':0.6,'į':0.6,'c':0.6}
LOGP = {c: math.log(DAZNIS.get(c, 0.05) / 100) for c in ABC}

def vizeneris(t, raktas, dec=False):
    # Raktas taikomas tik abėcėlės raidėms; ne abėcėlės simboliams praleidžiamas.
    k = [IDX[x] for x in raktas.lower()]; r = []; j = 0
    for c in t:
        if c in IDX:
            p = k[j % len(k)]
            r.append((ABC if c.islower() else UP)[(IDX[c] + (-p if dec else p)) % N]); j += 1
        else:
            r.append(c)
    return "".join(r)

def rasti_rakta(sifras):
    s = [IDX[c] for c in sifras if c in IDX]                  # tik raidžių srautas
    def ic(m):                                               # vid. sutapimo indeksas
        v = []
        for c in range(m):
            seka = s[c::m]; n = len(seka)
            if n > 1:
                cnt = [seka.count(x) for x in set(seka)]
                v.append(sum(a*(a-1) for a in cnt) / (n*(n-1)))
        return sum(v) / len(v)
    m = max(range(1, 25), key=ic)                            # rakto ilgis
    poslinkis = lambda seka: max(range(N), key=lambda sh: sum(LOGP[ABC[(x-sh) % N]] for x in seka))
    raktas = "".join(ABC[poslinkis(s[c::m])] for c in range(m))
    for d in range(1, len(raktas)):                          # sutrumpinam, jei kartojasi
        if len(raktas) % d == 0 and raktas[:d] * (len(raktas)//d) == raktas:
            return raktas[:d]
    return raktas

def spresti(v):
    eil = Path(f"variantai/{v}.txt").read_text(encoding="utf-8").splitlines()
    idx = lambda z: next(i for i, l in enumerate(eil) if l.strip() == z)
    i1, i2 = idx("1."), idx("2.")
    blokas = [l for l in eil[i1+1:i2] if l.strip()]          # 1 dalis: šifras + raktas
    at1 = vizeneris(blokas[0].strip(), blokas[1].strip(), dec=True)
    raktas = rasti_rakta("\n".join(eil[i2+1:]))              # 2 dalis: rasti raktą
    return (f"Variantas: {v}\n\n1. Atšifruotas tekstas:\n   {at1}\n\n"
            f"2. Rastas raktas: {raktas}\n")

if __name__ == "__main__":
    print(spresti(int(sys.argv[1])))
